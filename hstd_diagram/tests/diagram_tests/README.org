#!/usr/bin/env python
import math
from itertools import combinations

nodes = {
    "center": (105, 91.603),
    "a": (155, 178.21),
    "c": (5, 91.603),
    "d": (155, 5),
    "b": (205, 264.81),
}

edges = [("center", "a"), ("center", "c"), ("center", "d"), ("a", "b")]

def dist(p1, p2):
    return math.hypot(p1[0] - p2[0], p1[1] - p2[1])

print("Pairwise distances:")
for n1, n2 in combinations(nodes, 2):
    print(f"  {n1} - {n2}: {dist(nodes[n1], nodes[n2]):.4f}")

print("\nEdge lengths:")
for n1, n2 in edges:
    print(f"  {n1} -> {n2}: {dist(nodes[n1], nodes[n2]):.4f}")
